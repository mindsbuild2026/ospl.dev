-- Create a strictly IMMUTABLE wrapper for converting a text array to string
-- This resolves the "generation expression is not immutable" error on search_vector.
create or replace function immutable_array_to_string(text[], text) 
returns text 
language sql immutable strict parallel safe 
as $$
  select array_to_string($1, $2);
$$;

create table if not exists prompts (
  id uuid primary key default gen_random_uuid(),
  slug text not null unique,
  title text not null,
  short_description text not null,
  description text not null,
  category_id uuid not null references categories(id),
  subcategory_id uuid references subcategories(id),
  author_id uuid not null references authors(id) on delete restrict,
  author_user_id uuid,
  prompt_type_id uuid references prompt_types(id),
  difficulty text,
  license_type text not null default 'MIT',
  commercial_use boolean not null default true,
  attribution_required boolean not null default false,
  featured boolean not null default false,
  verified boolean not null default false,
  community_validated boolean not null default false,
  current_version text not null default '1.0.0',
  meta_title text,
  meta_description text,
  seo_keywords text[],
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  moderation_status text not null default 'pending',
  moderated_by uuid,
  moderated_at timestamptz,
  moderated_notes text,
  search_vector tsvector generated always as (
    setweight(to_tsvector('english'::regconfig, coalesce(title, '')), 'A') ||
    setweight(to_tsvector('english'::regconfig, coalesce(short_description, '')), 'B') ||
    setweight(to_tsvector('english'::regconfig, coalesce(description, '')), 'C') ||
    setweight(to_tsvector('english'::regconfig, coalesce(immutable_array_to_string(seo_keywords, ' '), '')), 'D')
  ) stored
);

create view category_summaries as
select
  c.id,
  c.slug,
  c.name,
  c.description,
  c.icon_name,
  c.is_trending,
  c.seo_h1,
  c.meta_title,
  c.meta_description,
  c.sort_order,
  count(p.id)::int as prompt_count
from categories c
left join prompts p on p.category_id = c.id
group by c.id
order by c.sort_order;

create view collection_summaries as
select
  c.id,
  c.slug,
  c.name,
  c.description,
  c.icon_name,
  c.category_id,
  c.sort_order,
  count(pc.prompt_id)::int as prompt_count
from collections c
left join prompt_collections pc on pc.collection_id = c.id
group by c.id
order by prompt_count desc;

create view author_summaries as
select
  a.id,
  a.handle,
  a.name,
  a.avatar_url,
  a.verified,
  a.reputation,
  count(p.id)::int as prompt_count
from authors a
left join prompts p on p.author_id = a.id
group by a.id
order by a.reputation desc;

left join prompt_types pt on pt.id = p.prompt_type_id;

--
-- Search RPC
--
create or replace function search_prompt_cards(
  search_text text,
  category_slug_filter text,
  category_id_filter uuid,
  tag_filters text[],
  sort_key text default 'Trending',
  limit_count int default 60,
  offset_count int default 0
)
returns setof prompt_card_view
language plpgsql stable
as $$
begin
  return query
  select *
  from prompt_card_view
  where
    (search_text is null or trim(search_text) = '' or
      to_tsvector('english'::regconfig,
        coalesce(title,'') || ' ' ||
        coalesce(short_description,'') || ' ' ||
        coalesce(category_name,'') || ' ' ||
        coalesce(subcategory_name,'') || ' ' ||
        coalesce(immutable_array_to_string(tags, ' '),'')
      ) @@ plainto_tsquery('english'::regconfig, search_text))
    and (category_slug_filter is null or trim(category_slug_filter) = '' or category_slug_filter = category_name)
    and (category_id_filter is null or category_id_filter = category_id)
    and (
      tag_filters is null
      or array_length(tag_filters, 1) = 0
      or (tags is not null and tags @> tag_filters)
    )
  order by
    case when sort_key = 'Trending' then trending_score end desc,
    case when sort_key = 'Most Popular' then popularity_rank end asc,
    case when sort_key = 'Most Copied' then copies end desc,
    case when sort_key = 'Most Viewed' then views end desc,
    case when sort_key = 'Highest Rated' then rating end desc,
    case when sort_key = 'Most Bookmarked' then bookmarks end desc,
    case when sort_key = 'Most Discussed' then comments end desc,
    case when sort_key = 'Newest' then created_at end desc,
    case when sort_key = 'Recently Updated' then updated_at end desc,
    case when sort_key = 'A-Z' then title end asc,
    trending_score desc
  limit limit_count offset offset_count;
end;
$$;

create or replace function increment_prompt_view(prompt_id_input uuid)
returns void
language plpgsql volatile
as $$
begin
  insert into prompt_events(prompt_id, event_type, created_at)
  values (prompt_id_input, 'view', now());
  update prompt_metrics
  set views = prompt_metrics.views + 1,
      updated_at = now()
  where prompt_id = prompt_id_input;
end;
$$;

create or replace function increment_prompt_bookmark(prompt_id_input uuid, delta_input int)
returns void
language plpgsql volatile
as $$
begin
  insert into prompt_events(prompt_id, event_type, event_metadata, created_at)
  values (prompt_id_input, 'bookmark', jsonb_build_object('delta', delta_input), now());
  update prompt_metrics
  set bookmarks = prompt_metrics.bookmarks + delta_input,
      updated_at = now()
  where prompt_id = prompt_id_input;
end;
$$;

create or replace function create_prompt_from_payload(payload jsonb)
returns uuid
language plpgsql volatile
security definer
set search_path = public, pg_catalog
as $$
declare
  prompt_id uuid;
  category_uuid uuid;
  author_uuid uuid;
  prompt_type_uuid uuid;
  slug_text text := trim(coalesce(payload->>'slug',''));
  title_text text := trim(coalesce(payload->>'title',''));
begin
  if slug_text = '' then
    raise exception 'create_prompt_from_payload: slug is required';
  end if;
  if title_text = '' then
    raise exception 'create_prompt_from_payload: title is required';
  end if;

  category_uuid := nullif(trim(coalesce(payload->>'categoryId','')),'')::uuid;
  author_uuid := nullif(trim(coalesce(payload->>'authorId','')),'')::uuid;
  prompt_type_uuid := nullif(trim(coalesce(payload->>'promptTypeId','')),'')::uuid;

  if category_uuid is null then
    raise exception 'create_prompt_from_payload: categoryId is required';
  end if;
  if author_uuid is null then
    raise exception 'create_prompt_from_payload: authorId is required';
  end if;

  insert into prompts (
    slug, title, short_description, description, category_id, subcategory_id, author_id,
    author_user_id, prompt_type_id, difficulty, license_type, commercial_use,
    attribution_required, featured, verified, community_validated,
    current_version, meta_title, meta_description, seo_keywords,
    moderation_status, created_at, updated_at
  )
  values (
    slug_text,
    title_text,
    coalesce(payload->>'shortDescription',''),
    coalesce(payload->>'description',''),
    category_uuid,
    nullif(trim(coalesce(payload->>'subCategoryId','')),'')::uuid,
    author_uuid,
    nullif(trim(coalesce(payload->>'authorUserId','')),'')::uuid,
    prompt_type_uuid,
    nullif(trim(coalesce(payload->>'difficulty','')),''),
    coalesce(payload->>'licenseType','MIT'),
    coalesce((coalesce(payload->>'commercialUse','false'))::boolean, false),
    coalesce((coalesce(payload->>'attributionRequired','false'))::boolean, false),
    coalesce((coalesce(payload->>'featured','false'))::boolean, false),
    coalesce((coalesce(payload->>'verified','false'))::boolean, false),
    coalesce((coalesce(payload->>'communityValidated','false'))::boolean, false),
    coalesce(payload->>'currentVersion','1.0.0'),
    nullif(trim(coalesce(payload->>'metaTitle','')),''),
    nullif(trim(coalesce(payload->>'metaDescription','')),''),
    array(select jsonb_array_elements_text(coalesce(payload->'seoKeywords', '[]'::jsonb))),
    coalesce(payload->>'moderationStatus','pending'),
    now(),
    now()
  )
  returning id into prompt_id;

  insert into prompt_metrics(prompt_id) values (prompt_id);
  return prompt_id;
end;
$$;

-- Row Level Security (RLS)

alter table authors enable row level security;

create policy author_public_select on authors for select using (true);

create policy author_insert_self on authors for insert with check (
  auth.role() = 'admin' or auth.uid() = user_id
);

create policy author_update_self on authors for update using (
  auth.role() = 'admin' or auth.uid() = user_id
);

alter table prompts enable row level security;

create policy prompt_public_select on prompts for select using (true);

create policy prompt_insert_authors on prompts for insert with check (
  auth.role() = 'admin'
  or (
    auth.role() = 'authenticated'
    and (
      author_user_id = auth.uid()
      or exists (
        select 1 from authors a where a.id = author_id and a.user_id = auth.uid()
      )
    )
  )
);

create policy prompt_update_author on prompts for update using (
  auth.role() = 'admin'
  or (
    auth.role() = 'authenticated'
    and (
      author_user_id = auth.uid()
      or exists (
        select 1 from authors a where a.id = prompts.author_id and a.user_id = auth.uid()
      )
    )
  )
);

create policy prompt_delete_admin on prompts for delete using (
  auth.role() = 'admin'
);
